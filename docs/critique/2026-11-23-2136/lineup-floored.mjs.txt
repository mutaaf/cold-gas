// The Lineup by keys with Assist when ↑ is simply held ("Hold ↑ to go", src/guide.js:25) all the way to the line:
// Space pressed at z PRESS (the cue "Space at the line" shows from z −14, src/main.js:1527), held LIFT s (↑ released
// at the press), ← 0.8 s, then Shift to the ground. Does any timing land?
import { createRun, stepRun } from '/home/user/cold-gas/src/sim/run.js';
import { newInput } from '/home/user/cold-gas/src/sim/vehicle.js';
import { scoreRun } from '/home/user/cold-gas/src/sim/judge.js';
import { initPhysics } from '/home/user/cold-gas/src/sim/engine.js';
await initPhysics();
let ok = 0, n = 0;
for (const press of [-14, -10, -7, -4, -2, 0]) for (const lift of [0.3, 0.4, 0.6, 0.8]) {
  const run = createRun('lineup'); let tL = null, tR = null, vAt = 0, apex = 0;
  while (run.phase !== 'done' && run.tick < 240 * 40) {
    const c = run.car, inp = newInput(); inp.assist = 1; const now = run.tick / 240;
    inp.steer = Math.max(-1, Math.min(1, c.pos.x * 0.25));
    if (tL === null && c.pos.z >= press) { tL = now; vAt = Math.hypot(c.vel.x, c.vel.z); }
    if (tL === null) inp.throttle = 1;
    else if (now - tL < lift) inp.lift = 1;
    else if (run.phase === 'air') { if (tR === null) tR = now; if (now - tR < 0.8) inp.roll = -1; else inp.level = 1; }
    else inp.brake = 1;
    stepRun(run, inp); apex = Math.max(apex, c.pos.y);
  }
  const s = scoreRun(run.t, { assist: true }); const landed = run.t.settledUpright && run.t.touchTick > 0; n++; if (landed && s.verdict !== 'Wrecked') ok++;
  console.log(JSON.stringify({ press, lift, vAtPress: +vAt.toFixed(1), apex: +apex.toFixed(1), verdict: s.verdict, total: s.total, end: run.t.endReason }));
}
console.log(`${ok}/${n} landed without a wreck`);
