// Production, a first visit on a phone (390x844 touch, coldgas_howto='new'), three runs through the page's own Retry ([data-again]).
// Run 1 and 3: one finger does what the pad says (260 ms reaction). Run 2: lets go as told, then never touches again (hands off).
// Logs pad words, when the beat's score is on screen (sim s after touchdown and wall s), the card, the sheet's coach text.
import { chromium } from '/home/user/cold-gas/node_modules/@playwright/test/index.mjs';
const [,, OUT = 'shots', URL = 'https://mutaafaziz.com/coldgas/'] = process.argv;
const browser = await chromium.launch({ executablePath: '/opt/pw-browsers/chromium-1243/chrome-linux64/chrome', args: ['--use-gl=angle', '--use-angle=swiftshader', '--enable-unsafe-swiftshader'] });
const c = await browser.newContext({ viewport: { width: 390, height: 844 }, deviceScaleFactor: 1, isMobile: true, hasTouch: true });
await c.addInitScript(() => { if (!sessionStorage.getItem('fv')) { localStorage.setItem('coldgas_howto', 'new'); sessionStorage.setItem('fv', '1'); } });
await c.route('**/boxoffice/**', (r) => r.fulfill({ status: 200, body: '' }));
await c.route('**/api/scores', (r) => r.request().method() === 'POST' ? r.fulfill({ status: 204 }) : r.continue());
await c.route('**/api/obs', (r) => r.fulfill({ status: 204 }));
await c.route(/googlesyndication|doubleclick|googleads|adservice|google\.com\/(pagead|recaptcha)/, (r) => r.abort());
const p = await c.newPage(); const errs = []; p.on('pageerror', (e) => errs.push(e.message)); p.on('console', (m) => { if (m.type() === 'error') errs.push('console: ' + m.text()); });
const T0 = Date.now(); const log = (...a) => console.error(((Date.now() - T0) / 1000).toFixed(1), ...a);
const shot = (n) => p.screenshot({ path: `${OUT}/${n}.png` });
await p.goto(URL);
await p.waitForFunction(() => window.__coldgas?.S.mode === 'title', null, { timeout: 120000 });
await p.waitForTimeout(2500);
const title = await p.evaluate(() => ({ first: document.documentElement.hasAttribute('data-first-title'), scheme: window.__coldgas.touch.get?.().scheme, course: window.__coldgas.stage.course }));
log('title', JSON.stringify(title)); await shot('01-title');
const cdp = await c.newCDPSession(p);
const sleep = (ms) => new Promise((r) => setTimeout(r, ms));
const results = [];
for (let run = 1; run <= 3; run++) {
  const tTap = Date.now();
  if (run === 1) await p.locator('[data-start]').tap();
  else { const b = p.locator('[data-again]:visible').first(); log('retry visible', await b.count()); await b.tap(); }
  await p.waitForFunction(() => window.__coldgas.S.mode === 'live', null, { timeout: 90000 });
  const tLive = Date.now(); log(`run ${run} live after`, (tLive - tTap) / 1000, 's');
  await shot(`r${run}-02-live`);
  const pad = await p.locator('[data-onethumb]').boundingBox().catch(() => null);
  const padC = pad && { x: pad.x + pad.width / 2, y: pad.y + pad.height / 2 };
  let down = false, tapped = false; const seen = new Set(); const shotAt = {};
  const set = async (on) => { if (on === down) return; down = on; await cdp.send('Input.dispatchTouchEvent', { type: on ? 'touchStart' : 'touchEnd', touchPoints: on ? [padC] : [] }); };
  let simTouch = null, beat = null;
  while (Date.now() - tLive < 150000) {
    const st = await p.evaluate(() => { const S = window.__coldgas.S, r = S.run; const b = document.querySelector('[data-beat]'); return { mode: S.mode, phase: r?.phase, t: r?.time ?? r?.t?.time, tick: r?.tick, y: r?.car.pos.y, z: r?.car.pos.z, inv: r?.t.invertedTime, verb: document.querySelector('[data-thumb-verb]')?.textContent || '', act: document.querySelector('[data-thumb-act]')?.textContent || '', cue: document.querySelector('[data-cue].on')?.textContent || '', beat: !!b && !b.hidden, total: document.querySelector('[data-beat-total]')?.textContent || '' }; });
    if (st.mode !== 'live') { log('mode', st.mode); break; }
    const key = st.verb + '|' + st.act + '|' + st.cue; if (!seen.has(key)) { seen.add(key); log('pad', key, st.phase, st.y?.toFixed(1), st.z?.toFixed(1), 'tick', st.tick); }
    if (!padC) break;
    const handsOff = run === 2 && tapped;
    if (/let go/i.test(st.act) && !tapped) { await set(false); await sleep(260); await set(true); tapped = true; if (run === 2) { await sleep(200); await set(false); } }
    else if (!handsOff && /^hold/i.test(st.verb)) { if (!down) await sleep(260); await set(true); }
    const m = (n, cond) => { if (cond && !shotAt[n]) { shotAt[n] = 1; return shot(`r${run}-${n}`); } };
    await m('03-runup', st.phase === 'drive' && st.z > -4); await m('04-climb', st.phase === 'ramp' && st.y > 8); await m('05-hang', st.inv > 0.4);
    await m('06-release', st.phase === 'air' && st.y > 15 && tapped); await m('07-carry', st.phase === 'air' && st.z < -16); await m('08-rollout', st.phase === 'air' && st.y < 6);
    if ((st.phase === 'landed' || st.phase === 'crashed') && simTouch == null) { simTouch = st.tick; log('touch', st.phase, 'tick', st.tick); await shot(`r${run}-09-touchdown`); }
    if (st.beat && /\d/.test(st.total) && !beat) { beat = { simAfter: simTouch != null ? (st.tick - simTouch) / 240 : null, total: st.total }; log('beat score on screen', JSON.stringify(beat)); await shot(`r${run}-10-beat`); }
    await sleep(40);
  }
  await set(false);
  await p.waitForFunction(() => window.__coldgas.S.mode === 'results', null, { timeout: 90000 }).catch(() => log('no results'));
  await p.waitForTimeout(3000); await shot(`r${run}-11-sheet`);
  const card = await p.evaluate(() => { const S = window.__coldgas.S, c = S.card; const sc = document.querySelector('[data-screen="results"]');
    return c && { total: c.total, verdict: c.verdict, crashed: c.facts.crashed, sink: c.facts.sink, padMiss: c.facts.padMiss, clearance: c.facts.clearance, invertedS: c.facts.invertedS, apex: c.facts.apex, gas: c.facts.gasLeft ?? c.facts.gas, judges: c.judges?.map((j) => j.name + ' ' + j.score),
      sheet: sc ? sc.innerText.replace(/\s+/g, ' ').slice(0, 700) : null, againVisible: !!document.querySelector('[data-again]:not(.quick-retry)')?.offsetParent }; });
  card.beat = beat; card.tapToLive = (tLive - tTap) / 1000;
  log('card', JSON.stringify(card)); results.push(card);
}
console.log(JSON.stringify({ title, results, errs }, null, 1));
await browser.close();
