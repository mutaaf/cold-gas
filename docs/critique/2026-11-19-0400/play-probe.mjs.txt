async function fly(label, opts) {
  await p.evaluate((o) => { const g = window.__coldgas; g.S.autopilot = g.createAutopilot(g.stage.course, o || {}); g.skipCountdown(); }, opts);
  const seen = {}, log = [];
  const marks = [['climb', (s) => s.phase === 'ramp' && s.prog > 0.4], ['hang', (s) => s.upY < -0.5], ['release', (s) => s.phase === 'air'], ['carry', (s) => s.phase === 'air' && s.vy < -2], ['rollout', (s) => s.phase === 'air' && s.upY > 0.9 && s.y < 12], ['touchdown', (s) => s.phase === 'landed' || s.phase === 'done' || s.crashed], ['after', (s) => s.mode !== 'live']];
  for (let i = 0; i < 1400; i++) {
    const s = await p.evaluate(() => { const g = window.__coldgas, S = g.S; for (let k = 0; k < 3 && S.mode === 'live'; k++) g.liveStep(1 / 60); const r = S.run, c = r?.car; const up = c?.q ? 1 - 2 * (c.q.x * c.q.x + c.q.z * c.q.z) : 1;
      return { mode: S.mode, phase: r?.phase, y: c?.pos.y, z: c?.pos.z, vy: c?.vel.y, upY: up, prog: c ? (r.progress ?? 0) : 0, cue: document.querySelector('[data-cue]')?.textContent || '', crashed: !!r?.t?.crashed }; });
    if (s.prog === 0 && s.phase === 'ramp') s.prog = 0.5;
    if (s.cue && log[log.length - 1] !== s.cue) log.push(s.cue);
    for (const [n, f] of marks) if (!seen[n] && f(s)) { seen[n] = { y: +(s.y ?? 0).toFixed(1), z: +(s.z ?? 0).toFixed(1) }; await p.evaluate(() => new Promise((r) => requestAnimationFrame(() => requestAnimationFrame(r)))); await shot(`${label}-${n}`); }
    if (s.mode !== 'live') break;
  }
  await p.waitForTimeout(9000);
  const card = await p.evaluate(() => { const S = window.__coldgas.S, k = S.card; return k && { total: k.total, verdict: k.verdict, sink: k.facts?.sink, clearance: k.facts?.clearance, apex: k.facts?.apex ?? k.facts?.height, airPct: k.facts?.airPct, brakeG: k.facts?.brakeG, padMiss: k.facts?.padMiss, crashed: k.facts?.crashed, tAir: S.run?.t && (S.run.t.touchTick - S.run.t.releaseTick) / 240 }; });
  const sheet = await p.evaluate(() => document.querySelector('[data-screen="results"]')?.innerText.replace(/\s+/g, ' ').slice(0, 600));
  await shot(`${label}-results`);
  return { seen, cues: log.slice(0, 20), card, sheet };
}
