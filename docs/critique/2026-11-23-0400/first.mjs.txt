// Production, a first visit on a phone (390x844 touch, coldgas_howto='new'): the title as it comes, a real tap on its Jump,
// one finger on whatever pad the game shows doing what the pad says (a 260 ms reaction), screenshots of each moment,
// then the landing beat, the sheet, a Retry tap and a second run. Wall-clock times of the core loop.
import { chromium } from '/home/user/cold-gas/node_modules/@playwright/test/index.mjs';
const [,, OUT = 'shots', URL = 'https://mutaafaziz.com/coldgas/', RUNS = '2'] = process.argv;
const browser = await chromium.launch({ executablePath: '/opt/pw-browsers/chromium-1243/chrome-linux64/chrome', args: ['--use-gl=angle', '--use-angle=swiftshader', '--enable-unsafe-swiftshader'] });
const c = await browser.newContext({ viewport: { width: 390, height: 844 }, deviceScaleFactor: 1, isMobile: true, hasTouch: true });
await c.addInitScript(() => { if (!sessionStorage.getItem('fv')) { localStorage.setItem('coldgas_howto', 'new'); sessionStorage.setItem('fv', '1'); } });
await c.route('**/boxoffice/**', (r) => r.fulfill({ status: 200, body: '' }));
await c.route('**/api/scores', (r) => r.request().method() === 'POST' ? r.fulfill({ status: 204 }) : r.continue());
await c.route('**/api/obs', (r) => r.fulfill({ status: 204 }));
await c.route(/googlesyndication|doubleclick|googleads|adservice|google\.com\/(pagead|recaptcha)/, (r) => r.abort());
const p = await c.newPage(); const errs = []; p.on('pageerror', (e) => errs.push(e.message)); p.on('console', (m) => { if (m.type() === 'error') errs.push('console: ' + m.text()); });
const T0 = Date.now(); const log = (...a) => console.error(((Date.now() - T0) / 1000).toFixed(1), ...a);
const shot = (n) => p.screenshot({ path: `${OUT}/${n}.png` });
await p.goto(URL);
await p.waitForFunction(() => window.__coldgas?.S.mode === 'title', null, { timeout: 120000 });
log('title mode'); await p.waitForTimeout(2500);
const title = await p.evaluate(() => ({ first: document.documentElement.hasAttribute('data-first-title'), start: document.querySelector('[data-start]')?.textContent?.trim(), scheme: window.__coldgas.touch.get?.().scheme, course: window.__coldgas.stage.course,
  visibleButtons: [...document.querySelectorAll('[data-screen="title"] button, [data-screen="title"] a')].filter((b) => b.offsetParent && b.getBoundingClientRect().height > 0).map((b) => b.textContent.trim().replace(/\s+/g, ' ').slice(0, 40)) }));
log('title', JSON.stringify(title)); await shot('01-title');
const cdp = await c.newCDPSession(p);
const sleep = (ms) => new Promise((r) => setTimeout(r, ms));
const results = [];
for (let run = 1; run <= +RUNS; run++) {
  const tTap = Date.now();
  if (run === 1) await p.locator('[data-start]').tap();
  else { const b = p.locator('[data-r-retry], [data-retry]').first(); await b.tap(); }
  await p.waitForFunction(() => ['countdown', 'live'].includes(window.__coldgas.S.mode) || document.querySelector('#howto:not([hidden]), [data-howto]:not([hidden])'), null, { timeout: 60000 });
  const after = await p.evaluate(() => window.__coldgas.S.mode); log('after tap', after); await shot(`r${run}-02-after-tap`);
  await p.waitForFunction(() => window.__coldgas.S.mode === 'live', null, { timeout: 60000 });
  const tLive = Date.now(); log('live after', (tLive - tTap) / 1000, 's');
  const pad = await p.locator('[data-onethumb]').boundingBox().catch(() => null);
  const padC = pad && { x: pad.x + pad.width / 2, y: pad.y + pad.height / 2 };
  let down = false, tapped = false; const seen = new Set(); const shotAt = {};
  const set = async (on) => { if (on === down) return; down = on; await cdp.send('Input.dispatchTouchEvent', { type: on ? 'touchStart' : 'touchEnd', touchPoints: on ? [padC] : [] }); };
  let tTouch = null;
  while (Date.now() - tLive < 120000) {
    const st = await p.evaluate(() => { const S = window.__coldgas.S, r = S.run; return { mode: S.mode, phase: r?.phase, y: r?.car.pos.y, z: r?.car.pos.z, inv: r?.t.invertedTime, verb: document.querySelector('[data-thumb-verb]')?.textContent || '', act: document.querySelector('[data-thumb-act]')?.textContent || '', cue: document.querySelector('[data-cue].on')?.textContent || '' }; });
    if (st.mode !== 'live') break;
    const key = st.verb + '|' + st.act; if (!seen.has(key)) { seen.add(key); log('pad', key, 'cue', st.cue, st.phase, st.y?.toFixed(1), st.z?.toFixed(1)); }
    if (!padC) break;
    if (/let go/i.test(st.act) && !tapped) { await set(false); await sleep(260); await set(true); tapped = true; }
    else if (/^hold/i.test(st.verb)) { if (!down) await sleep(260); await set(true); }
    const m = (n, cond) => { if (cond && !shotAt[n]) { shotAt[n] = 1; return shot(`r${run}-${n}`); } };
    if (run === 1) { await m('03-runup', st.phase === 'drive' && st.z > -4); await m('04-climb', st.phase === 'ramp' && st.y > 8); await m('05-hang', st.inv > 0.4); await m('06-release', st.phase === 'air' && st.y > 15 && tapped); await m('07-carry', st.phase === 'air' && st.z < -16); await m('08-rollout', st.phase === 'air' && st.y < 6); }
    if (st.phase === 'landed' && !tTouch) { tTouch = Date.now(); if (run === 1) await shot('r1-09-touchdown'); }
    await sleep(40);
  }
  await set(false);
  // The beat: when the score is first readable after touchdown.
  const tBeat0 = Date.now();
  let scoreSeen = null;
  for (let i = 0; i < 300 && !scoreSeen; i++) { const s = await p.evaluate(() => { const b = document.querySelector('[data-beat]'); const sc = document.querySelector('[data-beat] [data-beat-score], [data-r-total], [data-total]'); return { mode: window.__coldgas.S.mode, beat: !!b && !b.hidden, txt: (sc?.offsetParent && sc.textContent.trim()) || '' }; }); if (/\d/.test(s.txt)) scoreSeen = { ...s, t: (Date.now() - (tTouch || tBeat0)) / 1000 }; else await sleep(100); }
  log('score readable', JSON.stringify(scoreSeen));
  if (run === 1) await shot('r1-10-beat');
  await p.waitForFunction(() => window.__coldgas.S.mode === 'results', null, { timeout: 60000 }).catch(() => {});
  const tRes = Date.now(); await p.waitForTimeout(2500); await shot(`r${run}-11-sheet`);
  const card = await p.evaluate(() => { const S = window.__coldgas.S, c = S.card; const vis = (sel) => { const e = document.querySelector(sel); return e && e.offsetParent ? e.textContent.trim().replace(/\s+/g, ' ').slice(0, 200) : null; };
    return c && { total: c.total, verdict: c.verdict, crashed: c.facts.crashed, sink: c.facts.sink, padMiss: c.facts.padMiss, clearance: c.facts.clearance, invertedS: c.facts.invertedS, tier: c.tier,
      calm: document.querySelector('[data-screen="results"]')?.hasAttribute('data-calm'), sheetText: vis('[data-screen="results"]'),
      retry: !!document.querySelector('[data-r-retry], [data-retry]')?.offsetParent }; });
  card.touchToSheet = tTouch ? (tRes - tTouch) / 1000 : null; card.tapToLive = (tLive - tTap) / 1000;
  log('card', JSON.stringify(card)); results.push(card);
}
console.log(JSON.stringify({ title, results, errs }, null, 1));
await browser.close();
