    const d = t[a] - o[a]; if (Math.abs(d) < 1e-9) { if (o[a] < b0 || o[a] > b1) return false; continue; }
    let u0 = (b0 - o[a]) / d, u1 = (b1 - o[a]) / d; if (u0 > u1) [u0, u1] = [u1, u0]; lo = Math.max(lo, u0); hi = Math.min(hi, u1); if (lo > hi) return false; }
