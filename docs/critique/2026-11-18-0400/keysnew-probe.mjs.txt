import { createRun, stepRun, wallProgress } from '/home/user/cold-gas/src/sim/run.js';
import { newInput } from '/home/user/cold-gas/src/sim/vehicle.js';
import { scoreRun } from '/home/user/cold-gas/src/sim/judge.js';
import { coach } from '/home/user/cold-gas/src/sim/coach.js';
import { initPhysics } from '/home/user/cold-gas/src/sim/engine.js';
await initPhysics();
const course = process.argv[2] || 'show';
for (const upAir of [false, true]) for (const upGround of [false, true]) {
  let ok = 0, n = 0; const why = {}, sinks = [];
  for (const releaseAt of [0.9, 0.95, 1.0]) for (const boost of [0, 0.3, 0.6]) for (const shiftAt of [Infinity, 0.5, 1.5, 2.5]) {
    const run = createRun(course); let released = false, airT = 0, boostT = 0; n++;
    while (run.phase !== 'done') {
      const c = run.car, inp = newInput(); inp.assist = 1;
      inp.steer = Math.max(-1, Math.min(1, c.pos.x * 0.25));
      if (run.phase === 'drive') { inp.throttle = 1; inp.fan = 1; }
      else if (run.phase === 'ramp' && !released) { inp.throttle = 1; inp.fan = 1; if (wallProgress(c.pos) >= releaseAt) released = true; if (released) inp.lift = 1; }
      else if (run.phase === 'ramp' || run.phase === 'air') {
        airT += 1 / 240; if (upAir) inp.pitch = 1;
        if (boostT < boost) { inp.thrust = 1; boostT += 1 / 240; }
        if (airT > shiftAt) inp.level = 1;
      } else { if (upGround) inp.throttle = 1; else inp.brake = 1; }
      stepRun(run, inp);
    }
    const s = scoreRun(run.t, { assist: true }), t = run.t;
    const good = t.settledUpright && !t.landedOnRamp && !t.fellOff && t.touchTick > 0 && !t.contact;
    sinks.push(t.sink);
    if (good) ok++; else { const r = coach(run.log ?? run.inputLog, course, s); const id = (r?.items ?? r)?.[0]?.id ?? '?'; why[id] = (why[id] || 0) + 1; }
  }
  sinks.sort((a, b) => a - b);
  console.log(course, `upAir=${upAir} upGround=${upGround}`.padEnd(30), `${ok}/${n}`, 'median sink', sinks[n >> 1].toFixed(1), JSON.stringify(why));
}
