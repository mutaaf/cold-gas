import { initPhysics } from '../src/sim/engine.js';
import { flyKeysLineup } from './lib/daily-reach.mjs';
import { scoreRun } from '../src/sim/judge.js';
await initPhysics();
// A fixed player: lag 0.25 s, early 0, the cue's roll and LAND; vary run-up speed and the Space press.
const seq = (vals) => { let i = 0; const f = () => vals[i++]; f.pick = () => vals[i++]; return f; };
for (const speed of [6, 7, 8, Infinity]) for (const hold of [0.25, 0.3, 0.35, 0.4, 0.45, 0.5, 0.6]) {
  // draws: lag, speed, early, hold, side, land, steer
  const run = flyKeysLineup(seq([(0.25 - 0.18) / 0.25, speed, 0.375, hold, 0.2, 'cue', 0.7]));
  const t = run.t, s = scoreRun(t, { assist: true });
  console.log(`speed ${String(speed).padEnd(8)} hold ${hold.toFixed(2)}  peak ${t.maxHeight?.toFixed?.(1)} clear ${t.minClearance?.toFixed?.(2)} contact ${t.contact} ${s.verdict.padEnd(14)} ${s.total} landZ ${t.landZ?.toFixed?.(1)} launchV ${t.launchSpeed?.toFixed?.(1) ?? ''}`);
}
